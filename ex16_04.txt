// 음성 제어(→자전거, 버스, 비행기, 보트, 택시, 지하철)로 아이콘 표시, 랜덤 이동, 음성 출력

let speech = new p5.Speech(); 			// 음성합성 객체 생성
let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let words = ['자전거', '버스', '비행기', '보트', '택시', '지하철'];
let traffic = ['eb29', 'e530', 'e539', 'e532', 'e559', 'e56f'];	// Code Point
let index = -1, preIndex = -1, x = 0, y = 0, tx = 0.0, ty = 0.5, speaking = false;

function setup() {
  createCanvas(500, 400);
  textFont('Material Symbols Outlined');		// 폰트 설정
  speech.setLang('ko-KR'); 		// 음성합성 언어 설정(한국어)
  speech.started(speechStart);		// 음성합성이 시작되면 콜백함수 speechStart() 호출
  speech.ended(speechEnd);		// 음성합성이 끝나면 콜백함수 speechEnd() 호출
  rec.onEnd = recEnd;			// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, false);			// true: 연속음성인식, false: 최종 인식 결과만 출력
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 자전거, 버스, 비행기, 보트, 택시, 지하철', 20, 30);
  text('인식결과: ' + words[index], 20, 60);
  if (index >=0 && index < traffic.length) {	// 바르게 인식한 경우
    textSize(100);
    let icon = char(unhex(traffic[index]));	// 16진수→10진수→문자형으로 형변환
    text(icon, x, y);				// 아이콘 문자 표시
    x = noise(tx) * width;			// x값 범위: 0∼width
    y = noise(ty) * height;			// y값 범위: 0∼height
    tx += 0.01;
    ty += 0.01;
  } else {					// 오인식의 경우
    textSize(40);
    text('키워드를 말해주세요', 70, height/2);
  }
}

function recResult() {				// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString; 			// 음성인식 결과를 result에 저장
  console.log(result);
  if (result.indexOf('자전거') !== -1) {		// 인식 문장 중 '자전거'가 포함된 경우
    index = 0;
  } else if (result.indexOf('버스') !== -1) {	// 인식 문장 중 '버스'가 포함된 경우
    index = 1;
  } else if (result.indexOf('비행기') !== -1) {  	// 인식 문장 중 '비행기'가 포함된 경우
    index = 2;
  } else if (result.indexOf('보트') !== -1) {  	// 인식 문장 중 '보트'가 포함된 경우
    index = 3;
  } else if (result.indexOf('택시') !== -1) {  	// 인식 문장 중 '택시'가 포함된 경우
    index = 4;
  } else if (result.indexOf('지하철') !== -1) {  	// 인식 문장 중 '지하철'이 포함된 경우
    index = 5;
  } else {					// 오인식의 경우
    index = -1;
  }
  if (index >= 0 && index !== preIndex) {	// 합성음성 한 번만 출력
    speech.speak(words[index]);
  }
  preIndex = index;
}

function speechStart() {		// 콜백함수: 음성합성 시작될 때 자동 호출
  console.log('음성인식 중지(음성합성 시작)');
  speaking = true;
  rec.stop();				// 음성인식 중지 → recEnd() 호출
}

function speechEnd() {			// 콜백함수: 음성합성 종료 시 자동 호출
  console.log('음성인식 시작(음성합성 종료)');
  speaking = false;
  rec.start();				// 음성인식 재시작
}

function recEnd() {			// 콜백함수: 시간초과로 음성인식 종료 시 자동 호출
  if (!speaking) {			// 음성합성이 종료된 경우에만 참
    console.log('음성인식 재시작');
    rec.start();				// 음성인식 재시작
  }
}
