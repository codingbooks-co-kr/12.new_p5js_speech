// p5.js: 음성 제어(→왼쪽, 오른쪽, 위쪽, 아래쪽, 정지)로 원의 x축 및 y축 위치값(0∼255) 송신

let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let port = new p5.WebSerial();				// 시리얼 통신을 위한 객체 생성
let result = '';
let value1 = 0, value2 = 0;
let x = 200, y = 200, dx = 0, dy = 0;

function setup() {
  createCanvas(400, 400);
  port.getPorts();			// 포트 사용 가능 여부 확인
  port.on('noport', makePortButton);	// 기존 선택한 포트가 없으면 makePortButton() 호출
  port.on('portavailable', openPort);	// 사용 가능한 포트가 있으면 openPort() 자동 호출
  rec.onEnd = recEnd; 	// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, true);		// true: 연속음성인식, true: 중간 인식 결과도 출력
}

function draw() {
  background(0);
  noStroke();
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 왼쪽, 오른쪽, 위쪽, 아래쪽, 정지', 20, 30);
  text('각도: ' + nf(value1, 3), 20, 60);
  text('밝기: ' + nf(value2, 3), 20, 90);
  
  if (result.indexOf('왼쪽') !== -1) {
    dx = -1;
  } else if (result.indexOf('오른쪽') !== -1) {
    dx = 1;
  } else if (result.indexOf('위쪽') !== -1) {
    dy = -1;
  } else if (result.indexOf('아래쪽') !== -1) {
    dy = 1;
  } else if (result.indexOf('정지') !== -1) {
    dx = 0;
    dy = 0;
  }
  x += dx;
  y += dy;
  x = constrain(x, 0, width);		// x좌표값을 0∼width로 범위 제한
  y = constrain(y, 0, height);		// y좌표값을 0∼height로 범위 제한
  fill(255, 0, 0, 200);
  ellipse(x, y, 50, 50);
  
  value1 = int(map(x, 0, width, 0, 180));	// 서보모터 각도값을 0∼180으로 변환 
  value2 = int(map(y, 0, height, 0, 255));	// LED 밝기값을 0∼255로 변환
  port.write('A');     			// 동기신호 송신
  port.write(value1);			// 서보모터 각도값 송신
  port.write(value2);			// LED 밝기값 송신
}

function recResult() {			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  result = rec.resultString;		// 음성인식 결과를 result에 저장
  console.log(result);
}

function recEnd() {			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start();				// 음성인식 재시작
}

function makePortButton() {			// 포트 첫 연결 시 포트 선택 버튼 생성
  let portButton = createButton('포트 선택');	// 버튼 생성
  portButton.mousePressed(choosePort);	// 버튼 누르면 choosePort() 자동 호출
  function choosePort() {			// 콜백함수
    port.requestPort();				// 시리얼 포트 선택창 오픈
  }
}

function openPort() {				// 콜백함수
  port.open().then(initiateSerial);		// 포트를 오픈 후 initiateSerial() 자동 호출
  function initiateSerial() {			// 콜백함수
    console.log('포트 오픈');
  }
}
