// 음성인식으로 인식된 결과를 연속으로 텍스트 입력창에 자동 입력

let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let textArea, words = '';

function setup() {
  noCanvas();
  createElement('h2', '음성인식으로 문장 입력');	// h2의 크기로 제목 출력
  textArea = createElement('textarea');		// 텍스트 입력창 생성
  textArea.style('font-size', '20px');		// 텍스트 입력창의 폰트 크기 설정
  textArea.attribute('rows', '10');		// 텍스트 입력창의 행 수: 10행
  textArea.attribute('cols', '40');			// 텍스트 입력창의 열 수: 40열

  rec.onEnd = recEnd; 			// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, false); 				// 연속음성인식 설정, 최종 인식 결과만 출력
}

function draw() {
  textArea.value(words);			// 텍스트 입력창에 텍스트(words) 써넣기
}

function recResult() {				// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString; 			// 음성인식 결과를 result에 저장
  words += result;				// 인식된 텍스트를 이전 텍스트와 연결
  words += '. ';				// 최종 연결된 텍스트에 마침표 첨부
}

function recEnd() {				// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start(); 					// 음성인식 재시작
}
