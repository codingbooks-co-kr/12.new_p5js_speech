// “객체 리터럴”의 속성인 key 및 value의 개수 및 값 추출

function setup() {
  let traffic = { 자전거: 'eb29', 버스: 'e530', 비행기: 'e539' };	// 객체 리터럴
  let obj1 = Object.keys(traffic);		// ❶객체의 속성 중 key를 배열로 반환
  console.log('개수: ' + obj1.length);		// ❷keys 배열의 길이: 3
  console.log(obj1);				// key 배열: ["자전거", "버스", "비행기"]
  
  let obj2 = Object.values(traffic);		// ❸객체의 속성 중 value를 배열로 반환
  console.log('개수: ' + obj2.length);		// ❹values 배열의 길이: 3
  console.log(obj2);				// values 배열: ["eb29", "e530", "e539"]
  
  for(let key in traffic) {			// ❺“for in” 반복문으로 key와 value 추출
    console.log(key, traffic[key]);
  }
}
