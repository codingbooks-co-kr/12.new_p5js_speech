// p5.js: 인터럽트(interrupt) 방식
// 푸시버튼값 수신하여 원 색상 변경

let speech = new p5.Speech();
let port = new p5.WebSerial();	
let values, preValues;

function setup() {
  createCanvas(400, 400);
  port.getPorts();
  port.on('noport', makePortButton);
  port.on('portavailable', openPort);
  port.on('data', serialEvent);
  speech.setLang('ko-KR');
}

function draw() {
  background(0);
  if (values === 0) {
    fill(255, 255, 0);
    if (preValues === 1) {
      speech.speak('버튼을 눌렀어요');
    }
  } else if (values === 1) {
    fill(255, 0, 0);
  }
  ellipse(width/2, height/2, 200, 200);
  preValues = values;
}

// 수신값이 있으면 serialEvent() 자동 호출
function serialEvent() {
  values = port.read();
}

function makePortButton() {
  let portButton = createButton('포트선택');
  portButton.mousePressed(choosePort);
  function choosePort() {
    port.requestPort();
  }
}

function openPort() {
  port.open().then(initiateSerial);
  function initiateSerial() {
    console.log('포트 오픈');
  }
}
