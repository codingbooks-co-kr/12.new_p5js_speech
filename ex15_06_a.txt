// 아두이노: 가변저항값 송신

#define VARIABLE_R A1

void setup() {
  pinMode(VARIABLE_R, INPUT);
  Serial.begin(9600);
  while (Serial.available() == 0) {	// 수신 버퍼에 수신값이 없는 동안 반복 실행
    Serial.write('A');			// 시리얼 통신 시작신호 전송
    delay(300);				// 0.3초 대기
  }
}

void loop( ) { }

void serialEvent() {				// 시리얼 버퍼에 값이 수신되면 자동 호출
  int signal = Serial.read();			// 수신: 통신제어신호('B') 수신
  int val = analogRead(VARIABLE_R);		// 가변저항값(0∼1023) 읽기
  int value = map(val, 0, 1023, 0, 255);	// 가변저항값을 1바이트 범위(0∼255)로 변환
  Serial.write(value);				// 송신: 가변저항값 전송
}
