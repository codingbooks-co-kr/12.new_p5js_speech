// 음성으로 좌표(x, y)를 말하면 해당 좌표에 원이 생성

let rec = new p5.SpeechRec('ko-KR', recResult);
let x = 200, y = 200;

function setup() {
  createCanvas(400, 400);
  background(0);
  createP('인식단어: x좌표, y좌표');
  rec.onEnd = recEnd; 		// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, false); 			// true: 연속음성인식, false: 최종 인식 결과만 출력
}

function draw() {
  stroke(255, 255, 0);
  fill(255, 255, 0);
  ellipse(x, y, 30, 30);
}

function recResult() {  			// 음성인식 결과 출력
  let result = rec.resultString; 		// 음성인식 결과를 result에 저장
  console.log(result);
  let rule = /[^0-9]/g; 			// “숫자가 아닌 문자열”을 매칭하는 정규표현식
  result = result.replace(rule, ' '); 	// “숫자가 아닌 문자열”을 공백 문자(' ')로 치환
  let str = splitTokens(result, ' ');	// 여러 개의 공백으로 분리된 배열을 반환

  if (str.length != 2 || isNaN(str[0]) || isNaN(str[1])) {
    console.log('[두 개의 숫자를 말해주세요]');
    return;				// 더 이상 처리하지 않고 함수를 빠져나감
  }
  if (str[0] < 0 || str[0] > width || str[1] < 0 || str[1] > height) {
    console.log('[캔버스 범위를 넘었습니다]');
    return;				// 더 이상 처리하지 않고 함수를 빠져나감
  }
  x = int(str[0]);
  y = int(str[1]);
}

function recEnd() {  			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start(); 				// 음성인식 재시작
}
