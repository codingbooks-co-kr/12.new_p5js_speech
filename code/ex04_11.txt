// 마우스가 원 내부인지 외부인지에 따라 배경색 변경

let r = 100.0;

function setup() {
  createCanvas(300, 300);
}

function draw() {
  if (check()) {		// 함수 호출부
    background(255, 0, 0);
  } else {
    background(0);
  }
  ellipse(width/2, height/2, r*2, r*2);
}

function check() {		// 함수 본체
  let d = dist(width/2, height/2, mouseX, mouseY);
  if (d < r) {
    return true;		// 함수 반환값
  } else {
    return false;		// 함수 반환값
  }
}
