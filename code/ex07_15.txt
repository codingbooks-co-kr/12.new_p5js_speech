// 음성합성을 사용하여 방향 이모지를 마우스로 누르면 음성으로 출력

let speech = new p5.Speech();	// 음성합성 객체 생성
let words = ['위쪽', '아래쪽'];
let direction = ['👆🏿', '👇🏿']; 		// 상하 방향 이모지: 
let space = 100, x1 = 30, x2 = 180, y = 150;

function setup() {
  createCanvas(300, 200);
  background(200);
  textSize(space);

  speech.setLang('ko-KR');		// 음성합성 언어 설정(한국어)
  speech.started(speechStart);		// 음성합성이 시작되면 콜백함수 speechStart() 호출
  speech.ended(speechEnd);		// 음성합성이 끝나면 콜백함수 speechEnd() 호출
}

function draw() {
  text(direction[0], x1, y);
  text(direction[1], x2, y);
}

function mousePressed() {
  if (mouseX > x1 && mouseX < x1+space && mouseY > y-space && mouseY < y) {
    speech.speak(words[0]); 		// 합성음성 출력
  } else if (mouseX > x2 && mouseX < x2+space && mouseY > y-space && mouseY < y) {
    speech.speak(words[1]); 		// 합성음성 출력
  }
}

function speechStart() {		// 콜백함수: 음성합성이 시작되면 자동 호출
  background(255, 0, 0);
}

function speechEnd() {			// 콜백함수: 음성합성이 끝나면 자동 호출
  background(200);
}
