// 수신값(→0 또는 1)으로 LED 점멸 제어, 푸시버튼값(→0 또는 1) 송신

#define BUTTON 2
#define LED 6

void setup() {
  pinMode(BUTTON, INPUT);
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
  while (Serial.available() == 0) {	// 수신 버퍼에 수신값이 없는 동안 반복 실행
    Serial.write('A');			// 시리얼 통신 시작신호 전송
    delay(300);				// 0.3초 대기
  }
}

void loop() { }

void serialEvent() {			// 시리얼 버퍼에 값이 수신되면 자동 호출
  int value1 = Serial.read();		// 수신: 수신 데이터(→0 또는 1) 읽기
  digitalWrite(LED, value1);		// LED 점멸 제어
  int value2 = digitalRead(BUTTON);	// 푸시버튼값 읽기
  Serial.write(value2);			// 송신: 푸시버튼값 전송
}
