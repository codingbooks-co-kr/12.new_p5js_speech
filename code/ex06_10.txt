// 음성으로 좌우 이동하는 선의 현재 위치에 원 대신 사각형으로 변경

let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let move_x = 200, dx = 0;

function setup() {
  createCanvas(400, 200);
  rec.onEnd = recEnd;		// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, true);		// true: 연속음성인식, true: 중간 인식 결과도 출력
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 왼쪽, 오른쪽, 정지', 20, 30);
  
  stroke(255);
  line(move_x, 0, move_x, height);		// 선을 좌우로 이동
  
  for (let x = 0; x <= width; x += 50) {
    if (move_x > x-25 && move_x < x+25) {
      fill(255, 0, 0);
      rectMode(CENTER);
      rect(x, height/2, 50, 50);
    } else {
      fill(255); 
      ellipse(x, height/2, 50, 50);
    }
  }
  move_x += dx;				// x축 좌표 이동
  if (move_x < 0) move_x = width;		// 왼쪽 캔버스 경계를 넘어서는 경우
  if (move_x > width) move_x = 0;		// 오른쪽 캔버스 경계를 넘어서는 경우
}

function recResult() {				// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString;			// 음성인식 결과를 result에 저장
  console.log(result);
  if (result.indexOf('왼쪽') !== -1) { 		// 인식 문장 중 '왼쪽'이 포함된 경우
    dx = -1;
  } else if (result.indexOf('오른쪽') !== -1) { 	// 인식 문장 중 '오른쪽'이 포함된 경우
    dx = 1;
  } else if (result.indexOf('정지') !== -1) { 	// 인식 문장 중 '정지'가 포함된 경우
    dx = 0;
  }
}

function recEnd() {				// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start();					// 음성인식 재시작
}
