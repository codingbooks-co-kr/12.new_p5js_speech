// 아두이노: 수신된 두 개의 정수로 서보모터 각도 및 LED 밝기 제어

#include <Servo.h>			// Servo 클래스가 선언된 헤더파일 가져오기
#define LED 6
#define SERVO A0
Servo servo;				// Servo 클래스 기반 servo 객체의 선언 및 생성

void setup() {
  pinMode(LED, OUTPUT);
  pinMode(SERVO, OUTPUT);
  servo.attach(SERVO);			// SERVO핀(→A0)을 servo 객체에 연결
  Serial.begin(9600);
}

void loop() { }

void serialEvent() {			// 시리얼 버퍼에 값이 수신되면 자동 호출
  if (Serial.available() >= 3) {  		// 수신 버퍼에 3바이트 이상 있을 때
    int value = Serial.read();		// 1바이트 수신값 읽기
    if (value == 'A') {         		// 동기신호 확인
      int value1 = Serial.read(); 	// 1바이트 수신값 읽기
      int value2 = Serial.read(); 	// 1바이트 수신값 읽기
      servo.write(value1);		// 서보모터 제어
      analogWrite(LED, value2);	// LED 밝기 제어
    }
  }
}
