// 음성합성: 속성(volume, rate, pitch) 변경

let speech = new p5.Speech();			// 음성합성 객체 생성
let myLabel, myInput, myCheckbox, myButton, vSlider, rSlider, pSlider;

function setup() {
  createCanvas(400, 200);
  speech.setLang('ko-KR');				// 음성합성 언어 설정(한국어)
  background(200);

  myInput = createInput('안녕하세요');			// 입력창 생성
  myInput.position(20, 30);				// 입력창 위치 설정
  speech.speak(myInput.value());			// 합성음성 출력

  myButton = createButton('Text To Speech');		// 버튼 생성
  myButton.position(200, 30);				// 버튼 위치 설정
  myButton.mousePressed(textToSpeech);		// 버튼을 누르면 콜백함수 호출

  myCheckbox = createCheckbox('문장 교체', false); 	// 첵크박스 생성
  myCheckbox.position(20, 60); 			// 첵크박스 위치 설정
  myCheckbox.changed(changeText); 			// 첵크 변경 시 콜백함수 호출

  myLabel = createDiv('Volume');
  myLabel.position(20, 90);
  vSlider = createSlider(0, 100, 100);			// 볼륨(volume) 슬라이더 생성
  vSlider.position(80, 90);				// 슬라이더 위치 설정
  vSlider.mouseReleased(Volume);			// 슬라이더값 변경시 콜백함수 호출
  
  myLabel = createDiv('Rate');
  myLabel.position(20, 120);
  rSlider = createSlider(10, 200, 100);			// 속도(rate) 슬라이더 생성
  rSlider.position(80, 120);				// 슬라이더 위치 설정
  rSlider.mouseReleased(Rate);				// 슬라이더값 변경시 콜백함수 호출
  
  myLabel = createDiv('Pitch');
  myLabel.position(20, 150);
  pSlider = createSlider(1, 200, 100);			// 피치(pitch) 슬라이더 생성
  pSlider.position(80, 150);				// 슬라이더 위치 설정
  pSlider.mouseReleased(Pitch);			// 슬라이더값 변경시 콜백함수 호출
}

function Volume() {					// 콜백함수
  // 0.0(silent) ∼ 1.0 (default=1.0)
  speech.setVolume(vSlider.value() / 100);		// 합성음성의 볼륨 크기 설정
}

function Rate() {					// 콜백함수
  // 0.1(very slow) ∼ 2.0(very fast). default=1.0
  speech.setRate(rSlider.value() / 100);		// 합성음성의 속도 설정
}

function Pitch() {					// 콜백함수
  // 0.01(very low) ∼ 2.0(very high). default=1.0
  speech.setPitch(pSlider.value() / 100);		// 합성음성의 피치 크기 설정
}

function textToSpeech() {				// 콜백함수
  speech.speak(myInput.value());			// 합성음성 출력
}

function changeText() {					// 첵크 변경 시 자동 호출
  if (myCheckbox.checked()) {				// 첵크박스에 첵크한 경우
    myInput.value('반갑습니다');				// 입력창의 값(→문자열) 변경
  } else {   						// 첵크박스에 첵크하지 않은 경우
    myInput.value('안녕하세요');				// 입력창의 값(→문자열) 변경
  }
}
