// 숫자, 문자 및 문자열을 캔버스에 표시

function setup() {
  createCanvas(300, 250);  		// 캔버스 크기를 가로*세로=300*250픽셀로 설정
  background(200);			// 배경색을 200으로 설정

  textSize(30);            		// 문자 크기: 30픽셀
  text(7, 10, 50);        		// 숫자를 좌표 (10, 50)에 표현
  text('H', 10, 100);  		// 문자를 좌표 (10, 100)에 표현
  text('Hello, p5.js!', 10, 150);	// 문자열을 좌표 (10, 150)에 표현

  line(150, 0, 150, 300);		// 세로선 그리기
  line(0, 200, 300, 200);		// 가로선 그리기
  textAlign(CENTER, CENTER);	// 문자열 가운데 정렬
  text("Hello", 150, 200);		// 문자열 가운데 정렬
}
