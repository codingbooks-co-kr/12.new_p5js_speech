// 아두이노: 가변저항값 2개 및 푸시버튼값 송신

#define VARIABLE_R1 A1
#define VARIABLE_R2 A2
#define BUTTON 2

void setup() {
  pinMode(VARIABLE_R1, INPUT);
  pinMode(VARIABLE_R2, INPUT);
  pinMode(BUTTON, INPUT);
  Serial.begin(9600);
  while (Serial.available() == 0) {	// 수신 버퍼에 수신값이 없는 동안 반복 실행
    Serial.println('A');			// 시리얼 통신 시작신호 전송 ([심화15.1] 참조)
    delay(300);				// 0.3초 대기
  }
}

void loop() { }

void serialEvent() {              		// 시리얼 버퍼에 값이 수신되면 자동 호출
  int signal = Serial.read();         		// 수신: 제어신호('B') 수신
  int value1 = analogRead(VARIABLE_R1); 	// 첫 번째 가변저항값(0∼1023) 읽기
  int value2 = analogRead(VARIABLE_R2); 	// 두 번째 가변저항값(0∼1023) 읽기
  int value3 = digitalRead(BUTTON); 		// 송신문자열: 푸시버튼값(→0 또는 1)
  Serial.print(value1);             		// 송신문자열: 첫 번째 가변저항값
  Serial.print(",");              		// 송신문자열: 전송값 사이를 쉼표(,)로 구분
  Serial.print(value2);             		// 송신문자열: 두 번째 가변저항값
  Serial.print(",");              		// 송신문자열: 전송값 사이를 쉼표(,)로 구분
  Serial.println(value3);				// 송신문자열: 버튼값, 개행문자를 함께 전송
}
