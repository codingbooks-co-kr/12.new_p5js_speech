// 키워드(→랜덤, 시작, 정지)를 포함한 문장을 인식하여 100개의 원을 랜덤 이동/시작/정지 제어

let rec = new p5.SpeechRec('ko-KR', recResult);
let x = [], y = [], dx = [], dy = [];	// 원의 x,y 좌표 및 속도를 배열로 설정
let num = 100;

function setup() {
  createCanvas(400, 400);
  createP('인식단어: 랜덤, 시작, 정지');
  rec.onEnd = recEnd; 		// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, false); 			// true: 연속음성인식, false: 최종 인식 결과만 출력
  begin();				// 원들이 중심에서 분산
}

function draw() {
  background(0, 10);			// 페이딩 효과
  noStroke();				// 외곽선 제거

  for (let i = 0; i < num; i++) {
    fill(255);
    ellipse(x[i], y[i], 10, 10);
    if (x[i] < 0 || x[i] > width) {
      dx[i] *= -1;
    }
    if (y[i] < 0 || y[i] > height) {
      dy[i] *= -1;
    }
    x[i] += dx[i];
    y[i] += dy[i];
  }
}

function begin() {			// x,y좌표 및 속도의 초깃값 설정
  for (let i = 0; i < num; i++) {
    x[i] = width / 2; 			// x좌표 배열의 초깃값
    y[i] = height / 2; 			// y좌표 배열의 초깃값
    dx[i] = random(-3, 3); 		// x좌표 배열의 속도 초깃값 랜덤 설정
    dy[i] = random(-3, 3); 		// y좌표 배열의 속도 초깃값 랜덤 설정
  }
}

function recResult() {  			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString; 		// 음성인식 결과를 result에 저장
  console.log(result);
  if (result.indexOf('랜덤') !== -1) {
    for (let i = 0; i < num; i++) {
      dx[i] = random(-3, 3);
      dy[i] = random(-3, 3);
    }
  } else if (result.indexOf('시작') !== -1) {
    begin();				// 원들이 중심에서 다시 분산
  } else if (result.indexOf('정지') !== -1) {
    for (let i = 0; i < num; i++) {
      dx[i] = 0;
      dy[i] = 0;
    }
  }
}

function recEnd() {  			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start(); 				// 음성인식 재시작
}
