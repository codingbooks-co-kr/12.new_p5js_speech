// 웹 페이지의 화면을 마우스로 누르면 합성음성으로 출력 (캔버스 미사용)

let speech = new p5.Speech(); 	// 음성합성 객체 생성 및 초기화
let poem;

function setup() {
  noCanvas();
  poem = '나 보기가 역겨워 가실 때에는 말없이 고이 보내 드리오리다. 영변에 약산 진달래꽃 아름 따다 가실 길에 뿌리오리다. 가시는 걸음걸음 놓인 그 꽃을 사뿐히 즈려 밟고 가시옵소서. 나 보기가 역겨워 가실 때에는 죽어도 아니 눈물 흘리오리다.'
  speech.setLang('ko-KR');		// 음성합성 언어 설정(한국어)
  createP('[음성합성으로 문장 읽기]');
  createP(poem);
}

function mousePressed() {
  speech.speak(poem); 		// 텍스트를 합성음성으로 읽기
}
