let n2 = 12.345;
let s1 = "A";
let s2 = "p5.js";
let b1 = true;
let b2 = false;

  console.log("n2 = " + n2);
  console.log("s1 = " + s1);
  console.log("s2 = " + s2);

  console.log("b2 = " + b2);
