// 음성 제어(→자전거, 버스, 비행기, 보트, 택시, 지하철)로 아이콘 좌우 반사, LED/부저음 출력

let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let port = new p5.WebSerial();				// 시리얼 통신을 위한 객체 생성
let words = ['자전거', '버스', '비행기', '보트', '택시', '지하철'];
let traffic = ['eb29', 'e530', 'e539', 'e532', 'e559', 'e56f'];	// Code Point
let index = 0, x = 250, dx = 3, values = false, preValues = false;

function setup() {
  createCanvas(500, 200);
  textFont('Material Symbols Outlined');		// 폰트 설정
  port.getPorts();			// 포트 사용 가능 여부 확인
  port.on('noport', makePortButton);	// 기존 선택한 포트가 없으면 makePortButton() 호출
  port.on('portavailable', openPort);	// 사용 가능한 포트가 있으면 openPort() 자동 호출
  port.on('data', serialEvent);		// 데이터 수신 시 ‘data’ 이벤트 발생→콜백함수 호출
  rec.onEnd = recEnd;			// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, false);			// true: 연속음성인식, false: 최종 인식 결과만 출력
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('인식단어: 자전거, 버스, 비행기, 보트, 택시, 지하철', 20, 30);
  text('인식결과: ' + words[index], 20, 60);

  textSize(100);
  let icon = char(unhex(traffic[index]));	// 16진수→10진수→문자형으로 형변환
  text(icon, x, 150);			// 아이콘 문자 표시
  x += dx;
  if (x < 0 || x > width) {		// 좌우 경계에 도달하면 
    dx *= -1;				// 이동 방향 전환
    values = !values;			// true 또는 false로 토글 → serialEvent()
  }
}

function serialEvent() {			// 콜백함수: 시리얼버퍼에 수신값이 있으면 자동 호출
  let signal = port.read(); 		// 수신: 시작신호('A') 또는 제어신호('B')를 수신
  if (values === true && preValues  === false) {		// 반사하는 경우
    port.write(0);			// 송신: 0 전송
  } else if (values == false && preValues === true) {		// 반사하는 경우
    port.write(1);			// 송신: 1 전송
  } else {				// 반사하지 않는 경우
    port.write(2);			// 송신: 2 전송
  }
  preValues = values;
}

function recResult() {			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString; 		// 음성인식 결과를 result에 저장
  console.log(result);
  if (result.indexOf('자전거') !== -1) {		// 인식 문장 중 '자전거'가 포함된 경우
    index = 0;
  } else if (result.indexOf('버스') !== -1) {	// 인식 문장 중 '버스'가 포함된 경우
    index = 1;
  } else if (result.indexOf('비행기') !== -1) {  	// 인식 문장 중 '비행기'가 포함된 경우
    index = 2;
  } else if (result.indexOf('보트') !== -1) {  	// 인식 문장 중 '보트'가 포함된 경우
    index = 3;
  } else if (result.indexOf('택시') !== -1) {  	// 인식 문장 중 '택시'가 포함된 경우
    index = 4;
  } else if (result.indexOf('지하철') !== -1) {  	// 인식 문장 중 '지하철'이 포함된 경우
    index = 5;
  }
}

function recEnd() {				// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start();					// 음성인식 재시작
}

function makePortButton() {			// 포트 첫 연결 시 포트 선택 버튼 생성
  let portButton = createButton('포트 선택');	// 버튼 생성
  portButton.mousePressed(choosePort);	// 버튼 누르면 choosePort() 자동 호출
  function choosePort() {			// 콜백함수
    port.requestPort();				// 시리얼 포트 선택창 오픈
  }
}

function openPort() {				// 콜백함수
  port.open().then(initiateSerial);		// 포트를 오픈 후 initiateSerial() 자동 호출
  function initiateSerial() {			// 콜백함수
    console.log('포트 오픈');
  }
}
