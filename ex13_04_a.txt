// 아두이노: 수신된 문자열로 LED 점멸 제어

#define LED 6

void setup( ) {
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
}

void loop() { }

void serialEvent() {			// 시리얼 버퍼에 값이 수신되면 자동 호출
  String value = Serial.readStringUntil('\n');
  if (value.equals("on")) {		// 또는 if (value == "on")
    digitalWrite(LED, 1);
  } else if (value.equals("off")) {	// 또는 else if (value == "off")
    digitalWrite(LED, 0);
  }
}
