// 음성합성으로 한국어 문장을 출력하고 음성합성 리스트 표시4

let speech = new p5.Speech();		// 음성합성 객체 생성
let words = '안녕하세요';

function setup() {
  noCanvas();
  speech.setLang('ko-KR');			// 음성합성 언어 설정
  speech.setVoice('Google 한국의');		// 한국어 합성음성 설정
  speech.listVoices();				// 합성음성 리스트 출력
  speech.speak(words);				// 합성음성 출력
}

function mousePressed() {
  speech.speak(words);				// 합성음성 출력
  createP(words);				// 브라우저에 문자열 출력
}
