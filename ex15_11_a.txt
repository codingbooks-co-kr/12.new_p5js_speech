// 아두이노: 색상값 두 개(→Red, Green) 수신 후 RGB LED 색상 제어, 가변저항값 2개 송신

#define RedLED 11
#define GreenLED 5
#define BlueLED 3
#define VARIABLE_R1 A1
#define VARIABLE_R2 A2

void setup() {
  pinMode(RedLED, OUTPUT);
  pinMode(GreenLED, OUTPUT);
  pinMode(BlueLED, OUTPUT);
  pinMode(VARIABLE_R1, INPUT);
  pinMode(VARIABLE_R2, INPUT);
  Serial.begin(9600);
  while (Serial.available() == 0) {	// 수신 버퍼에 수신값이 없는 동안 반복 실행
    Serial.println('A');			// 시리얼 통신 시작신호 전송 ([심화15.1] 참조)
    delay(300);				// 0.3초 대기
  }
}

void loop() { }

void serialEvent() {			// 시리얼 버퍼에 값이 수신되면 자동 호출
  int value1 = Serial.parseInt();		// 수신 문자열 중 첫 번째 숫자문자 추출, 정수 저장
  int value2 = Serial.parseInt();		// 수신 문자열 중 두 번째 숫자문자 추출, 정수 저장
  analogWrite(RedLED, value1);
  analogWrite(GreenLED, value2);
  analogWrite(BlueLED, 255);

  int value3 = analogRead(VARIABLE_R1);	// 첫 번째 가변저항값(0∼1023) 읽기
  int value4 = analogRead(VARIABLE_R2);	// 두 번째 가변저항값(0∼1023) 읽기
  Serial.print(value3);
  Serial.print(",");			// 가변저항값 사이를 콤마(,)로 구분
  Serial.println(value4);			// 송신: “value3,value4\r\n”의 문자열로 묶어 송신
}
