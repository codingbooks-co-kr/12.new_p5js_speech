// 2차원 색상 배열을 사용한 원의 색상 변화

let c = [ 				// 2차원 색상 배열의 선언 및 초기화
  [0,   0,   0,   0,  0], 
  [0, 150, 150, 150, 0], 
  [0, 150, 255, 150, 0], 
  [0, 150, 150, 150, 0], 
  [0,   0,   0,   0,  0]
];

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  for (let y = 0; y < 5; y++) {
    for (let x = 0; x < 5; x++) {
      fill(c[y][x], 0, 255);		// 색상 배열의 적용
      ellipseMode(CORNER);		// 원의 왼쪽 상단 모서리를 기준으로 원을 그리는 모드
      ellipse(x * 60, y * 60, 60, 60);
    }
  }
}
