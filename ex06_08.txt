// 음성인식(영어)으로 원의 컬러 색상 변경

let rec = new p5.SpeechRec('en-US', recResult);	// 영어 설정('en-US')
let col = 'WHITE';					// 초기 원 색상

function setup() {
  createCanvas(400, 300);
  rec.onEnd = recEnd;			// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, false);			// true: 연속음성인식, false: 최종 인식 결과만 출력
  createP('색상: black, silver, gray, white, maroon, red, purple, fuchsia(magenta), green, lime, olive, yellow, navy, blue, teal, aqua(cyan)');
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  text('[색상을 영어로 말해봅시다]', 20, 30);
  text('색상: ' + col, 20, 60);

  fill(col);				// 색상 문자열로 색상 설정
  ellipse(width/2, height/2, 150, 150);
}

function recResult() {			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString;		// 음성인식 결과를 result에 저장
  col = result.toUpperCase();		// 음성인식 결과를 대문자로 변환
}

function recEnd() {			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start();				// 음성인식 재시작
}
