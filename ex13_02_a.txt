// 아두이노: 수신된 정수(→0,1,2,3,4)로 4개의 LED 제어

const int LED[] = {6, 7, 8, 12};

void setup() {
  for (int i = 0; i < 4; i++) {
    pinMode(LED[i], OUTPUT);
  }
  Serial.begin(9600);
}

void loop() { }

void serialEvent() {    			// 시리얼 버퍼에 값이 수신되면 자동 호출
  int value = Serial.read();		// 수신 데이터 1바이트 읽기
  for (int i = 0; i < 4; i++) {		// LED 전부 소등
    digitalWrite(LED[i], 0);
  }
  for (int i = 0; i < value; i++) {	// 수신된 데이터(1∼4)만큼 LED 점등
    digitalWrite(LED[i], 1);
  }
}
