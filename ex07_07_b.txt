// 연속음성인식 (캔버스 미사용)

let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화

function setup() {
  noCanvas();
  createDiv('[음성인식 시작]');
  rec.start(true, false);			// 연속음성인식 설정, 최종 인식 결과만 출력
  rec.onEnd = recEnd;			// 음성인식 종료 시 콜백함수 recEnd() 호출
}

function recResult() {			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  createDiv(rec.resultString);		// 음성인식 결과를 웹 페이지에 표시
}

function recEnd() {			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start();				// 음성인식 재시작
}
