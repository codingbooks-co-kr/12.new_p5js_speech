// 키워드(→왼쪽, 오른쪽, 위쪽, 아래쪽, 정지)를 포함한 문장을 인식하고 원의 이동/정지 제어

let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let x = 200, y = 200, dx = 0, dy = 0;

function setup() {
  createCanvas(400, 400);
  rec.onEnd = recEnd;			// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, true);			// true: 연속음성인식, true: 중간 인식 결과도 출력
}

function draw() {
  background(0, 10);			// 페이딩 효과
  noStroke();				// 외곽선 제거
  
  textSize(20);
  fill(255, 255, 0);
  text('인식단어: 왼쪽, 오른쪽, 위쪽, 아래쪽, 정지', 10, 30);

  fill(255);
  ellipse(x, y, 50, 50);
  x += dx;
  y += dy;
  if (x < 0) x = width;			// 왼쪽 캔버스 경계를 넘어서는 경우
  if (y < 0) y = height;		// 위쪽 캔버스 경계를 넘어서는 경우
  if (x > width) x = 0;			// 오른쪽 캔버스 경계를 넘어서는 경우
  if (y > height) y = 0;		// 아래쪽 캔버스 경계를 넘어서는 경우
}

function recResult() {			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString; 		// 음성인식 결과
  console.log(result);
  if (result.indexOf('왼쪽') !== -1) { 
    dx=-1;
    dy=0;
  } else if (result.indexOf('오른쪽') !== -1) { 
    dx=1;
    dy=0;
  } else if (result.indexOf('위쪽') !== -1) { 
    dx=0;
    dy=-1;
  } else if (result.indexOf('아래쪽') !== -1) { 
    dx=0;
    dy=1;
  } else if (result.indexOf('정지') !== -1) { 
    dx = 0;
    dy = 0;
  }
}

function recEnd() {  			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start(); 				// 음성인식 재시작
}
