// p5.js: 가변저항값 2개 수신하여 원을 상하좌우 이동하고 원의 위치를 음성으로 출력2

let speech = new p5.Speech();	// 음성합성 객체 생성
let port = new p5.WebSerial();		// 시리얼 통신을 위한 객체 생성
let value1 = 0, value2 = 0, num1 = 0, num2 = 0, preNum1 = 0, preNum2 = 0;

function setup() {
  createCanvas(400, 400);
  port.getPorts();			// 포트 사용 가능 여부 확인
  port.on('noport', makePortButton);	// 기존 선택한 포트가 없으면 makePortButton() 호출
  port.on('portavailable', openPort);	// 사용 가능한 포트가 있으면 openPort() 자동 호출
  port.on('data', serialEvent);		// 데이터 수신 시 ‘data’ 이벤트 발생→콜백함수 호출
  speech.setLang('ko-KR');		// 음성합성 언어 설정(한국어)
}

function draw() {
  background(0, 10);			// 페이딩 효과
  stroke(255, 255, 0);
  line(width/2, 0, width/2, height);	// 경계선
  line(0, height/2, width, height/2);	// 경계선

  noStroke();				// 외곽선 제거
  fill(255, 255, 0);
  ellipse(value1, value2, 50, 50);	// 두 가변저항값에 따라 원의 좌표 이동
}

function serialEvent() {			// 콜백함수: 시리얼 버퍼에 값이 수신되면 자동 호출
  let str = port.readLine();		// 개행까지의 수신 문자열 읽기
  if (!str) return;			// 입력 문자열(→str)이 없으면 함수 빠져나가기
  let s = trim(str);			// 입력 문자열(→str)의 시작과 끝에 붙은 공백 삭제
  let values = split(s, ',');		// 콤마로 구분된 문자열을 values 배열에 저장
  value1 = map(float(values[0]), 0, 1023, 0, width-1);
  value2 = map(float(values[1]), 0, 1023, 0, height-1);
  
  num1 = int(value1 / (width/2));	// 중앙 세로선을 기점으로 num1 = 0 또는 1
  num2 = int(value2 / (height/2));	// 중앙 가로선을 기점으로 num2 = 0 또는 1
  if (num1 != preNum1 || num2 != preNum2) {	// num1,num2 값이 바뀔 때 true
    if (num1 < 1 && num2 < 1) {
      speech.speak('왼쪽위');				// 합성음성 출력
    } else if (num1 < 2 && num2 < 1) {
      speech.speak('오른쪽위');				// 합성음성 출력
    } else if (num1 < 1 && num2 < 2) {
      speech.speak('왼쪽아래');				// 합성음성 출력
    } else if (num1 < 2 && num2 < 2) {
      speech.speak('오른쪽아래');			// 합성음성 출력
    }
  }
  preNum1 = num1;
  preNum2 = num2;
}

function makePortButton() {			// 포트 첫 연결 시 포트 선택 버튼 생성
  let portButton = createButton('포트 선택');	// 버튼 생성
  portButton.mousePressed(choosePort);	// 버튼 누르면 choosePort() 자동 호출
  function choosePort() {			// 콜백함수
    port.requestPort();				// 시리얼 포트 선택창 오픈
  }
}

function openPort() {				// 콜백함수
  port.open().then(initiateSerial);		// 포트를 오픈 후 initiateSerial() 자동 호출
  function initiateSerial() {			// 콜백함수
    console.log('포트 오픈');
  }
}
