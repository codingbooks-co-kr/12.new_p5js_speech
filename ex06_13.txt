// 1000개의 점의 상하좌우 및 랜덤 이동을 음성으로 제어

let rec = new p5.SpeechRec('ko-KR', recResult);
let x = [], y = [], c = []; 		// x,y: 좌표 배열, c: 색상 배열
let dx = 0, dy = 0, num = 1000;

function setup() {
  createCanvas(400, 400);
  for (let i = 0; i < num; i++) {
    // 배열 요소의 초깃값 설정
    x[i] = random(width); 		// x좌표 초기 위치 랜덤 설정
    y[i] = random(height); 		// y좌표 초기 위치 랜덤 설정
    c[i] = color(random(255), random(255), random(255));
  }
  createP('인식단어: 왼쪽, 오른쪽, 위쪽, 아래쪽, 랜덤, 정지');
  rec.onEnd = recEnd; 		// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, false); 			// true: 연속음성인식, false: 최종 인식 결과만 출력
}

function draw() {
  background(0, 10);			// 페이딩 효과
  for (let i = 0; i < num; i++) {
    stroke(c[i]); 			// 점의 색상 설정
    strokeWeight(5); 			// 점의 굵기 설정
    point(x[i], y[i]);
    x[i] += random(dx); 		// x좌표 랜덤값(0.0∼1.0) 누적, 방향성: 오른쪽
    y[i] += random(dy); 		// y좌표 랜덤값(0.0∼1.0) 누적, 방향성: 아래쪽

    if (x[i] < 0) x[i] = width;		// 왼쪽 캔버스 경계를 넘어서는 경우
    if (x[i] > width) x[i] = 0;		// 오른쪽 캔버스 경계를 넘어서는 경우
    if (y[i] < 0) y[i] = height;		// 위쪽 캔버스 경계를 넘어서는 경우
    if (y[i] > height) y[i] = 0;		// 아래쪽 캔버스 경계를 넘어서는 경우
  }
}

function recResult() {  			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString; 		// 음성인식 결과를 result에 저장
  console.log(result);
  if (result.indexOf('왼쪽') !== -1) {
    dx = -1;
    dy = 0;
  } else if (result.indexOf('오른쪽') !== -1) {
    dx = 1;
    dy = 0;
  } else if (result.indexOf('위쪽') !== -1) {
    dx = 0;
    dy = -1;
  } else if (result.indexOf('아래쪽') !== -1) {
    dx = 0;
    dy = 1;
  } else if (result.indexOf('랜덤') !== -1) {
    dx = random(-1, 1);
    dy = random(-1, 1);
  } else if (result.indexOf('정지') !== -1) {
    dx = 0;
    dy = 0;
  }
}

function recEnd() {  			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start(); 				// 음성인식 재시작
}
