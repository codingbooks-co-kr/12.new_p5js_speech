// p5.js: 가변저항값 2개 수신 후 사각형 이동, 음성인식으로 원 이동 및 원의 색상값 2개 송신

let speech = new p5.Speech();			// 음성합성 객체 생성
let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let port = new p5.WebSerial();				// 시리얼 통신을 위한 객체 생성
let result = '', x = 100, y = 100, dx = 0, dy = 0, value1 = 0, value2 = 0, value3 = 200, value4 = 200;

function setup() {
  createCanvas(400, 400);
  port.getPorts();			// 포트 사용 가능 여부 확인
  port.on('noport', makePortButton);	// 기존 선택한 포트가 없으면 makePortButton() 호출
  port.on('portavailable', openPort);	// 사용 가능한 포트가 있으면 openPort() 자동 호출
  port.on('data', serialEvent);		// 데이터 수신 시 ‘data’ 이벤트 발생→콜백함수 호출
  speech.setLang('ko-KR');		// 음성합성 언어 설정(한국어)
  rec.onEnd = recEnd;	 		// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, true);			// true: 연속음성인식, true: 중간 인식 결과도 출력
}

function draw() {
  background(0, 10);			// 페이딩 효과
  noStroke();				// 외곽선 제거

  if (result.indexOf('왼쪽') !== -1) {		// 왼쪽 음성인식
    dx = -1;
    dy = 0;
  } else if (result.indexOf('오른쪽') !== -1) {	// 오른쪽 음성인식
    dx = 1;
    dy = 0;
  } else if (result.indexOf('위쪽') !== -1) {	// 위쪽 음성인식
    dx = 0;
    dy = -1;
  } else if (result.indexOf('아래쪽') !== -1) {	// 아래쪽 음성인식
    dx = 0;
    dy = 1;
  } else if (result.indexOf('정지') !== -1) {	// 정지 음성인식
    dx = 0;
    dy = 0;
  }
  x += dx;					// 원의 x축 방향 이동
  y += dy;					// 원의 y축 방향 이동	
  x = constrain(x, 0, width);			// x값 범위 제한: 0∼width
  y = constrain(y, 0, height);			// y값 범위 제한: 0∼height
  value1 = int(map(x, 0, width, 0, 255));	// Red 색상값 범위 변경(0∼255)
  value2 = int(map(y, 0, height, 0, 255));	// Green 색상값 범위 변경(0∼255)

  fill(value1, value2, 255);			// 원의 Red, Green 색상 변화
  ellipse(x, y, 100, 100);			// 음성인식으로 원의 좌표 이동
  
  fill(255, 255, 0, 200);
  rect(value3, value4, 100, 100);		// 가변저항 2개로 사각형의 좌표 이동
}

function mousePressed() {			// 마우스를 누르면 현재 원의 좌표 읽기
  speech.speak(`현재 원의 x좌표는 ${x}이고 y좌표는 ${y}입니다`);
}

function serialEvent() {			// 콜백함수: 시리얼 버퍼에 값이 수신되면 자동 호출
  let str = port.readLine();		// 개행까지의 수신 문자열 읽기
  if (!str) return;			// 입력 문자열(→str)이 없으면 함수 빠져나가기
  let s = trim(str);			// 입력 문자열(→str)의 시작과 끝에 붙은 공백 삭제
  let values = split(s, ',');		// 콤마로 구분된 문자열을 values 배열에 저장
  port.write(value1 + ',' + value2 + '\n');	// 복수개의 값을 하나의 문자열로 송신

  if (values.length !== 2) return;	// 실행 초기, 통신시작신호('A')이면 함수 빠져나가기
  value3 = map(float(values[0]), 0, 1023, 0, width);
  value4 = map(float(values[1]), 0, 1023, 0, height);
}

function recResult() {			// 콜백함수: 음성인식 결과 반환 시 자동 호출
  result = rec.resultString;		// 음성인식 결과를 result에 저장
  console.log(result);
}

function recEnd() {			// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start();				// 음성인식 재시작
}

function makePortButton() {			// 포트 첫 연결 시 포트 선택 버튼 생성
  let portButton = createButton('포트 선택');	// 버튼 생성
  portButton.mousePressed(choosePort);	// 버튼 누르면 choosePort() 자동 호출
  function choosePort() {			// 콜백함수
    port.requestPort();				// 시리얼 포트 선택창 오픈
  }
}

function openPort() {				// 콜백함수
  port.open().then(initiateSerial);		// 포트를 오픈 후 initiateSerial() 자동 호출
  function initiateSerial() {			// 콜백함수
    console.log('포트 오픈');
  }
}
