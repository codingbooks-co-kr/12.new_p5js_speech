// 음성 제어(→랜덤, 정지)로 동물 이모지를 랜덤 선택하고 음성합성으로 이모지명 출력

let speech = new p5.Speech();			// 음성합성 객체 생성
let rec = new p5.SpeechRec('ko-KR', recResult);	// 음성인식 객체 생성 및 초기화
let words = ['원숭이', '고양이', '호랑이', '돼지', '낙타', '코끼리'];
let animals = ['🐵', '🐱', '🐯', '🐷', '🐪', '🐘'];	// 
let index = 0, stop = false, toggle = true;

function setup() {
  createCanvas(400, 200);
  speech.setLang('ko-KR'); 		// 음성합성 언어 설정(한국어)
  speech.started(speechStart);		// 음성합성이 시작되면 콜백함수 speechStart() 호출
  rec.onEnd = recEnd; 		// 음성인식 종료 시 콜백함수 recEnd() 호출
  rec.start(true, true); 			// true: 연속음성인식, true: 중간 인식 결과도 출력
}

function draw() {
  background(0);
  fill(255, 255, 0);
  textSize(20);
  textAlign(CENTER, CENTER);
  text('인식단어: 랜덤, 정지', width/2, 30);
  textSize(100);
  text(animals[index], width/2, height/2);

  if (stop) {
    noLoop();					// draw() 반복 처리 정지
  } else {
    index = int(random(animals.length));	// 동물 이모지 배열의 인덱스 랜덤 선택
  }
}

function recResult() {				// 콜백함수: 음성인식 결과 반환 시 자동 호출
  let result = rec.resultString; 			// 음성인식 결과를 result에 저장
  console.log(result);
  if (result.indexOf('정지') !== -1) {		// 인식 문장 중 '정지'가 포함된 경우
    stop = true;
    if (toggle) {				// toggle이 true일 때만 합성음성 출력
      speech.speak(words[index]);
      toggle = false;
    }
  } else if (result.indexOf('랜덤') !== -1) {	// 인식 문장 중 '랜덤'이 포함된 경우
    stop = false;
    toggle = true;
    loop();					// draw() 반복 처리
  }
}

function recEnd() { 				// 콜백함수: 음성인식 종료 시 자동 호출
  console.log('[음성인식 재시작]');
  rec.start(); 					// 음성인식 재시작
}

function speechStart() {			// 콜백함수: 음성합성이 시작되면 자동 호출
  textSize(20);
  text(words[index], width/2, 160);
}
